XML / HTML XPath

{
  "typeId": 3,
  "search": {
    "url": "https://site.com/search?wd=@keyword",
    "list": "//div[contains(@class,'item')]",
    "name": ".//h3/text()",
    "source": "./a/@href"
  },
  "detail": { "url": "https://site.com@sourceRaw" },
  "fields": {
    "vod_name": "//h1/text()",
    "vod_pic": "//img[@class='cover']/@src",
    "vod_content": "//div[@class='desc']//text()"
  },
  "play": {
    "groups": "//div[@class='line']",
    "from": "./@data-from",
    "episodes": ".//a",
    "episodeName": "./text()",
    "episodeUrl": "./@href"
  }
}

XPath 支持：//、.//、/、./、[@attr="x"]、[contains(@class,"x")]、[1]、[last()]、@attr、text()。
